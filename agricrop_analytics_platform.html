schema.sql (MySQL Relational Architecture)

Includes constraints, multi-column uniqueness, foreign keys, and yield integrity checks.

CREATE DATABASE IF NOT EXISTS agricrop_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE agricrop_db;

-- 1. Regions Entity
CREATE TABLE IF NOT EXISTS regions (
    region_id VARCHAR(50) PRIMARY KEY,
    region_name VARCHAR(100) NOT NULL,
    country VARCHAR(100) DEFAULT 'Global',
    climate_zone VARCHAR(100) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 2. Crops Entity
CREATE TABLE IF NOT EXISTS crops (
    crop_id VARCHAR(50) PRIMARY KEY,
    crop_name VARCHAR(100) NOT NULL,
    category ENUM('Cereal', 'Grain', 'Legume', 'Cash Crop', 'Vegetable', 'Fruit', 'Oilseed', 'Other') DEFAULT 'Cereal',
    scientific_name VARCHAR(150) DEFAULT NULL,
    typical_min_yield FLOAT DEFAULT 0.0,
    typical_max_yield FLOAT DEFAULT 20.0,
    yield_unit VARCHAR(30) DEFAULT 't/ha',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 3. Crop Records Entity
CREATE TABLE IF NOT EXISTS crop_records (
    record_id VARCHAR(64) PRIMARY KEY,
    crop_id VARCHAR(50) NOT NULL,
    region_id VARCHAR(50) NOT NULL,
    year INT NOT NULL,
    cultivated_area DOUBLE NOT NULL,
    area_unit VARCHAR(20) NOT NULL DEFAULT 'ha',
    production DOUBLE NOT NULL,
    production_unit VARCHAR(20) NOT NULL DEFAULT 'metric tons',
    calculated_yield DOUBLE NULL,
    yield_unit VARCHAR(30) NOT NULL DEFAULT 't/ha',
    rainfall_mm DOUBLE NULL,
    notes TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    CONSTRAINT fk_record_crop FOREIGN KEY (crop_id) 
        REFERENCES crops (crop_id) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_record_region FOREIGN KEY (region_id) 
        REFERENCES regions (region_id) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT chk_cultivated_area CHECK (cultivated_area >= 0),
    CONSTRAINT chk_production CHECK (production >= 0),
    CONSTRAINT uq_crop_region_year UNIQUE (crop_id, region_id, year)
) ENGINE=InnoDB;

CREATE INDEX idx_records_year ON crop_records (year);
CREATE INDEX idx_records_crop_year ON crop_records (crop_id, year);
CREATE INDEX idx_records_region_year ON crop_records (region_id, year);
CREATE INDEX idx_records_rainfall ON crop_records (rainfall_mm);
